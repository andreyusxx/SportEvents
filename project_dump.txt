public class Participant
{
    public int Id { get; set; }

    public bool IsActive { get; set; }

internal class Program
{
    internal static void Main()
    {
        Console.OutputEncoding = System.Text.Encoding.UTF8;
        Console.WriteLine("Генерація 100 000 записів...");

        var participants = new List<Participant>(100000);
        for (int i = 0; i < 100000; i++)
        {
            participants.Add(new Participant
            {
                Id = i,
                IsActive = i % 2 == 0,
                Duration = TimeSpan.FromSeconds(1500)
            });
        }

        var sw = new Stopwatch();

        // 1. Чистий код (LINQ)
        sw.Start();
        var activeAthletes = participants.Where(p => p.IsActive).ToList();
        double totalTimeClean = activeAthletes.Sum(p => p.Duration.TotalSeconds);
        sw.Stop();
        var cleanTime = sw.ElapsedMilliseconds;

        sw.Reset();

        // 2. Оптимізований код (for)
        sw.Start();
        double totalTimeOpt = 0;
        for (int i = 0; i < participants.Count; i++)
        {
            if (participants[i].IsActive)
            {
                totalTimeOpt += participants[i].Duration.TotalSeconds;
            }
        }

        sw.Stop();
        var optTime = sw.ElapsedMilliseconds;

        Console.WriteLine($"Чистий код (LINQ): {cleanTime} мс");
